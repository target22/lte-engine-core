```yaml
# `python -m` needs the repo root on sys.path, which `-w /app` provides.
#
# Pipeline, unchanged from the Bash original:
#   pre-flight -> image build -> batch ingest docs/ into the CAS repo ->
#   compile public + internal graphs in memory -> push Tier 1 (origin) ->
#   push Tier 3 (cas-remote) -> dashboard.
#
# RUN IT ON THE HOST, NOT IN THE CONTAINER:
#   cd /home/SRKH && python3 -m lte.cli.pipeline --config config/build_config.yaml
# This file orchestrates containers — build-image runs `docker build`, and
# three steps run `docker run`. Inside an ephemeral --rm container there is
# no docker daemon, no git identity for batch-ingest, and no SSH agent for
# either push. (--list and --dry-run appear to work in the container only
# because neither executes anything.)
#
# ---------------------------------------------------------------------
# WHY THERE IS NO `DOCKER_RUN` VARIABLE AND NO `eval`
# ---------------------------------------------------------------------
# An earlier revision of this file factored the container invocation into
#   DOCKER_RUN: 'docker run --rm -v "$PWD":/app ... "$DOCKER_IMAGE"'
# and each step ran `eval "$DOCKER_RUN" python scripts/...`.
#
# That was wrong, and it failed in production on a two-word author name.
# `eval` re-parses the command AFTER the shell has already expanded and
# de-quoted it, so quoting applied in the outer command is gone by the time
# eval runs:
#
#   --author-name "${GIT_AUTHOR_NAME:-$(git config user.name)}"
#     pass 1 -> --author-name Alan N.        (quotes consumed)
#     pass 2 -> argv: [--author-name] [Alan] [N.]
#     result -> batch_sync.py: error: unrecognized arguments: N.
#     (lte.cli.ingest would fail the same way if eval came back)
#
# Note the asymmetry that makes this hard to spot: values quoted INSIDE the
# DOCKER_RUN string (its own escaped "$CAS_REPO") survive the second parse,
# because their quotes are still literal text at that point. Values quoted
# in the step's own command do not. So the volume mounts looked correct
# while the author name silently split in two.
#
# The `docker run` prefix is therefore written out in each of the three
# container steps. That repeats one line three times, and in exchange the
# command that runs is exactly the command written here — one parse pass,
# no second interpretation layer, and `--dry-run` shows the real thing.
# Paths and names containing spaces survive correctly. Same lesson as the
# earlier execFile/argv fix in the VS Code extension: build the argument
# vector, do not re-interpret a string.
#
# ---------------------------------------------------------------------
# WHAT THE BASH SCRIPT DID THAT A STEP LIST CANNOT
# ---------------------------------------------------------------------
#   1. NO STATE FLOWS BETWEEN STEPS. Each step is its own subprocess, so
#      $AUTHOR_NAME, $ORIGIN_URL and $SYNC_SUMMARY_LINE cannot be computed
#      once and reused. Steps re-derive what they need inline, or read a
#      declared file artifact (SYNC_LOG). The filesystem is the only
#      channel; the runner itself stays stateless by design.
#
#   2. CONDITIONAL LOGIC NOW LIVES IN PYTHON. The original's ~35 lines of
#      pre-flight branching were briefly four YAML block scalars here,
#      which was the weakest part of the translation: no syntax checking,
#      no linting, no test seam. They are now scripts/core_pipeline/preflight.py, a
#      single step below. The remaining conditionals in this file are
#      tier1-commit's emptiness guard (three lines) and the dashboard
#      heredoc.
#
#   3. A BARE `git commit` FAILS WHEN THERE IS NOTHING TO COMMIT, so
#      halt_on_error would abort on every run where docs/ is unchanged.
#      The emptiness guard in tier1-commit is load-bearing.
#
#   4. THE DASHBOARD IS DEGRADED. It reads SYNC_LOG and the two compiled
#      artifacts rather than captured stdout; per-step timings come from
#      the runner's own summary table.

config_version: "1.0.0"
```

```yaml
# config/corpus_layout.yaml
#
# DECLARATIVE CORPUS LAYOUT. The only place that says where the corpus lives
# on disk, how its files are named, and what each compile target reads and
# writes. Compiled by lte/engine/layout.py (compile_layout) and loaded by
# lte/io/config_reader.py next to taxonomy.yaml and linter_rules.json.
#
# THIS FILE REPRODUCES THE PRE-LAYOUT CONVENTIONS EXACTLY:
#   docs/<scope>/<partition>/<tractate...>/<slug>[.<role>].<vi|en>.md
#   compile public   -> public/data/graph.json          (docs/public only)
#   compile internal -> public/internal-data/graph-internal.json (both scopes)
#   compile all      -> alias of internal
# A compile with this file is byte-for-byte identical in its vi/en payload to
# the compile before the layout existed (graph_hash unchanged).
#
# CROSS-FILE CONTRACTS, all checked at load time:
#   * every partition in taxonomy.yaml gets a directory here (default:
#     <scope path>/<partition id>); ids listed under `partitions` must exist
#     in taxonomy.yaml;
#   * every taxonomy `scope` value must be a scope id declared here;
#   * `default_role` must be one of linter_rules.json#document_roles;
#   * scope directories are pairwise disjoint, and so are partition
#     directories;
#   * a target with `exposure: public` may only read level-0 scopes.
#
# NOT CONFIGURED HERE (still pinned elsewhere; see the migration notes):
#   nginx.conf routes, docker-compose.yml single-file volume mounts,
#   src/layouts/app.js fetch paths. Change an `artifact` path only together
#   with those three.

layout_version: "1.0.0"

# Directory under the repository root that holds the corpus. "." = repo root.
root: "docs"

# Visibility classes. `path` is relative to `root` (default: the id).
# `level`: a [!ref-...] from a file in a lower-level scope to an anchor in a
# higher-level scope is a boundary violation. Level 0 is publishable.
scopes:
  - {id: public,  path: public,  level: 0}
  - {id: private, path: private, level: 10}
  - {id: corporate, path: corporate, level: 5}

# Per-partition directory overrides, keyed by taxonomy.yaml id.
#   path          relative to the partition's scope directory (default: id)
#   default_role  role for files whose name has no role segment
#   pairs_with    this partition's files join the documents of another
#                 partition in the same scope, matched by relative path
# Empty: every partition lives at <scope>/<id>/, as before.
partitions: {}

filenames:
  # mode "suffix": <slug>[.<role>].<locale><ext>, locale required.
  # mode "none":   <slug>[.<role>]<ext>, every file has locale `default`.
  # The first value is the default --lang and the first key in the artifact.
  locale:
    mode: suffix
    values: [vi, en]
  extensions: [".md"]
  # The optional role segment's values come from
  # linter_rules.json#document_roles; they are not repeated here.

# Globs over paths relative to `root`. `*` and `?` stay within one path
# segment; `**` spans segments (`**/x` also matches `x` at the root).
# A matching file is not a corpus file: never compiled, never ingested,
# never an error.
ignore: []

# Draft files under any directory with this name map to root:
#   drafts/public/core/x.en.md -> docs/public/core/x.en.md
staging:
  root: drafts

targets:
  - id: public
    scopes: [public]
    artifact: public/data/graph.json
    exposure: public
  - id: internal
    aliases: [all]
    scopes: [public, private]
    artifact: public/internal-data/graph-internal.json
    exposure: restricted
```
